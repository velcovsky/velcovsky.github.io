Render jpg


Prostředí: součást, sestava
  
  • vyrendruje aktuální pohled jako obrázek .jpg s bílím pozadím (nastaveno pro českou mutaci Inventoru)
  • při generování lze nastavit kvalitu renderu pomocí jednoduchého rozhraní v pravu dole
  • uloží s názvem číslo součásti+popis+aktuální čas (umožní uložit více obrázku bez nutnosti přepisovat název) 
Sub Main()
	Dim oView As Inventor.View
	oView = ThisApplication.ActiveView
	
	'white background
'ThisApplication.ColorSchemes.Item("Prezentace").Activate
ThisApplication.ColorSchemes.BackgroundType = 52737
'ThisApplication.ColorSchemes.BackgroundType = 52738

'Slozka ulozeni
'folderName = "C:\trans"
folderName = ThisDoc.Path

'cas
iProperties.Value("Custom", "Time Saved") = Now.ToString("HH:mm:ss")

'nazev ukladani
nazev= iProperties.Value("Project", "Part Number")&"-"& iProperties.Value("Project", "Description") & "(" & iProperties.Value("Custom", "Time Saved") & ")"


'nazev ukladanych xls - uprava lomitek atd.
txt1 = nazev
txt2 = Replace(txt1,"/","_")
txt3 = Replace(txt2,":","_")
edit_nazev = txt3
	
	oView.DisplayMode = DisplayModeEnum.kRealisticRendering
	
	' start ray tracing
	oView.RayTracing=  True 
	oView.RayTracingQuality = RayTracingQualityEnum.kHighRayTracingQuality
	
	Dim bStop As Boolean 
	
	Do 
		ThisApplication.UserInterfaceManager.DoEvents
		
		If (oView.RayTracingProgress = 1) Then
			bStop = True
			oView.SaveAsBitmapWithRayTracing (folderName & "\" & edit_nazev & ".jpg")
		End If
		
		If (oView.RayTracing = False) Then
				bStop = True 
		End If
	Loop Until (bStop)
	
	' turn off ray tracing
	oView.RayTracing = False 
	
		'gradient background
'ThisApplication.ColorSchemes.Item("Prezentace").Activate
'ThisApplication.ColorSchemes.BackgroundType = 52737
ThisApplication.ColorSchemes.BackgroundType = 52738
End Sub
