ČTENÍ DAT Z EXCELU


Prostředí: součást, sestava
  
  • Na základě vstupních dat (SK a Registrační číslo) najde příslušný řádek v tabulce excelu
  • z nalezeného řádku zapíše do i-vlastností další potřebné hodnoty ( Název a doplňkový kód)
  • vhodné pro ověření správnosti zapsaných čísel
  • vhodné číst a zapisovat hodnoty pomocí formuláře

InventorVb.DocumentUpdate()

'Zkountroluje jestli existují parametry SK, reg_cislo, nazev_1, doplnkovy_kod.

Dim oDoc As Document = ThisApplication.ActiveEditDocument
Dim oParas As UserParameters = oDoc.ComponentDefinition.Parameters.UserParameters
Dim oPara As UserParameter
'SK
Try
    oPara = oParas.Item("SK")
Catch
    oPara = oParas.AddByValue("SK", "", UnitsTypeEnum.kTextUnits)
End Try
'reg_cislo
Try
    oPara = oParas.Item("reg_cislo")
Catch
    oPara = oParas.AddByValue("reg_cislo", "", UnitsTypeEnum.kTextUnits)
End Try
'nazev_1
Try
    oPara = oParas.Item("nazev_1")
Catch
    oPara = oParas.AddByValue("nazev_1", "", UnitsTypeEnum.kTextUnits)
End Try
'doplnkovy_kod
Try
    oPara = oParas.Item("doplnkovy_kod")
Catch
    oPara = oParas.AddByValue("doplnkovy_kod", "", UnitsTypeEnum.kTextUnits)
End Try

'otevre formular 
iLogicForm.ShowGlobal("Helios")

'najde radek s hodnotama parametru SK a reg_cislo
i = GoExcel.FindRow("C:\trans\Průměrná cena a PRODEJNÍ cena.xls","Průměrná cena a PRODEJNÍ cena", "SK", "=", Parameter("SK"), "Reg. číslo", "=", Parameter("reg_cislo"))

'ukaze okno scislem radku
'MessageBox.Show(i.ToString(), "Title")

'Parameter("SK") = GoExcel.CurrentRowValue("SK")
'Parameter("reg_cislo") = GoExcel.CurrentRowValue("Reg. číslo")

'najde hodnoty nazev 1 a doplnovy kod a zapis je do tabulky parametru
Parameter("nazev_1") = GoExcel.CurrentRowValue("Název 1")
Parameter("doplnkovy_kod") = GoExcel.CurrentRowValue("Doplňkový kód")

' zapise hodnoty parametru SK, reg. cislo nazev 1 a doplnkovy kod do ivlastnosti
iProperties.Value("Custom", "SK") = Parameter("SK")
iProperties.Value("Custom", "Reg. číslo")=Parameter ("reg_cislo")
iProperties.Value("Custom", "Název 1")=Parameter ("nazev_1")
iProperties.Value("Custom", "Doplňkový kód") = Parameter("doplnkovy_kod")

'vytvori skladove cislo  - SK a reg_cislo
iProperties.Value("Project", "Stock Number")= iProperties.Value("Custom", "SK")&iProperties.Value("Custom", "Reg. číslo")

'zkontroluje jestli existuji pripadne vytvori ivlastnosti 001_nazev_vykresu, 002_nazev_produktu, 003_cislo_vykresu, 004_projekt.
customPropertySet = odoc.PropertySets.Item("Inventor User Defined Properties")

'001_nazev_vykresu---------------- find or create custom properties -------------
Try
    oTest = customPropertySet.Item("001_nazev_vykresu")
Catch
    customPropertySet.Add("", "001_nazev_vykresu")
End Try

If iProperties.Value("Custom", "001_nazev_vykresu") = "" Then
iProperties.Value("Custom", "001_nazev_vykresu")=iProperties.Value("Project", "Part Number")

End If 

'002_nazev_produktu---------------- find or create custom properties -------------
Try
    oTest = customPropertySet.Item("002_nazev_produktu")
Catch
    customPropertySet.Add("", "002_nazev_produktu")
End Try

If iProperties.Value("Custom", "002_nazev_produktu") = "" Then
iProperties.Value("Custom", "002_nazev_produktu")=iProperties.Value("Project", "Description")

End If 

'003_cislo_vykresu ---------------- find or create custom properties -------------
Try
    oTest = customPropertySet.Item("003_cislo_vykresu")
Catch
    customPropertySet.Add("", "003_cislo_vykresu")
End Try

'004_projekt ---------------- find or create custom properties -------------
Try
    oTest = customPropertySet.Item("004_projekt")
Catch
    customPropertySet.Add("", "004_projekt")
End Try

'cislo projektu automaticky (funguje pri zprávne strukture slozek)
oPath = ThisDoc.Path
'Odstranění cesty
FNamePos = InStrRev(oPath, "OP-", -1) 'počet znaků
Name = Right(oPath, Len(oPath) -FNamePos)
ShortName = Right(Name, Len(Name) -2)

'Odstranění 7 znaku zleva
ShortName1 = Right(ShortName, Len(ShortName) -7)
'Odstranění 7 znaku zleva
Project = Left(ShortName, Len(ShortName) - Len(ShortName1))
'MessageBox.Show(Project, "slozka")


'zapise hodnotu projekt do ivlastnosti
iProperties.Value("Custom", "projekt") = Project
