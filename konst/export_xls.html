 ' ___Start of ilogic___

'work only in drawing files
'ukladani xls - save XLS 
'musi mit prirazeno cislo vykresu a na prvnim listu kusovnik
'value 001_Zeichnungsnummer do not be empty and on Blatt:1 has to be BOM


'check if iproperties 001_Zeichnungsnummer exist
On Error Resume Next
Dim pSuggestedVendor 
pSuggestedVendor = iProperties.Value("Custom", "001_Zeichnungsnummer")

If Err.Number <> 0 Then
    pSuggestedVendor = Isempty
    Err.Clear
    'if not delete value ZEICHNUNGSNUMMER
    iProperties.Value("Custom", "ZEICHNUNGSNUMMER")=""
    
Else
'else revrite value ZEICHNUNGSNUMMER same as 001_Zeichnungsnummer
iProperties.Value("Custom", "ZEICHNUNGSNUMMER")=iProperties.Value("Custom", "001_Zeichnungsnummer")

Dim oDrawDoc As DrawingDocument
oDrawDoc = ThisApplication.ActiveDocument

'  kontrola cisla vykresu 
' check if 001_Zeichnungsnummer is not emty
If (iProperties.Value("Custom", "001_Zeichnungsnummer") IsNot "" ) Then

Dim oSheet1 As Sheet
oSheet1 = oDrawDoc.Sheets.Item(1)

'  kontrola jestli je na prvnim listu  kusovnik 
' check if  Blatt:1 has BOM
If oSheet1.PartsLists.Count = 1 Then 

Dim oDoc As Inventor.DrawingDocument
oDoc = ThisDoc.Document
 
Dim oSheet As Inventor.Sheet
'oSheet = oDoc.Sheets(1) ' first sheet
oSheet = oDoc.Sheets("blatt:1") ' sheet by name
 
' say there is a Partslist on the sheet.
Dim oPartslist As PartsList
oPartslist = oSheet.PartsLists(1)

'Slozka ulozeni
'folder location
folderName = "S:\vel\xls"
' folderName = ThisDoc.Path

'nazev ukladani
'file name
nazev=iProperties.Value("Custom", "001_Zeichnungsnummer")& "(" & iProperties.Value("Custom", "002_Benennung_de" ) &")_Copyright Vel"

'nazev ukladanych xls - uprava lomitek atd.
'editing  name
txt1 = nazev
txt2 = Replace(txt1,"/","_")
txt3 = Replace(txt2,"*","_")
edit_nazev = txt3

'uloz xls
' export the Partslist to Excel.
oPartslist.Export(folderName & "\" & edit_nazev & ".xls", PartsListFileFormatEnum.kMicrosoftExcel)

End If

End If 

End If

On Error GoTo 0

' ___End of ilogic___
