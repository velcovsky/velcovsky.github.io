Worksheet = ("C:\trans\vykresy.xlsx")
Sheetno = ("List1")
' Open excel 
GoExcel.Open(Worksheet, Sheetno)
' Define Range
RowStart = 2
RowEnd = 10000
For count = RowStart To RowEnd
	' If it's blank count it
	If String.IsNullOrEmpty(GoExcel.CellValue("A" & count)) Then 
		i = i + 1
	End If
Next
' Next empty row is max rows minus blank rows plus one
row = RowEnd - i + 1
' MsgBox("Inserting Data on Row " & row)
GoExcel.CellValue("A" & row) = iProperties.Value("Custom", "003_cislo_vykresu")
GoExcel.CellValue("B" & row) = iProperties.Value("Custom", "001_nazev_vykresu")
GoExcel.CellValue("C" & row) = iProperties.Value("Summary", "Author")
GoExcel.CellValue("D" & row) = DateString
GoExcel.CellValue("E" & row) = iProperties.Value("Project", "Stock Number")
GoExcel.Save

Else
' MsgBox("Value already exist")	
End If
